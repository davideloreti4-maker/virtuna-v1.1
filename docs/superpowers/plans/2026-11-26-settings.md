```ts
// src/lib/whop/membership.ts
export interface LiveMembership {
  /** Whop's status as sent: "active" | "trialing" | "past_due" | "canceled" | "expired" | "completed" | other. */
  status: string;
  cancelAtPeriodEnd: boolean;
  /** ISO time of the period's end, or null. Whop sends seconds or an ISO string (`renewal_period_end`). */
  renewsAt: string | null;
}
/** null: Whop has no such membership (404). Throws on any other failure (network, 5xx, missing key). */
export async function fetchWhopMembership(membershipId: string, fetchImpl?: typeof fetch): Promise<LiveMembership | null>;
/** active, trialing or past_due, and not cancel_at_period_end: Whop will bill again. */
export function membershipRenews(m: LiveMembership): boolean;

//   { membership: null }                       — no Whop membership on the caller's row (or Whop 404)
//   { membership: LiveMembership }
//   { unknown: true }                           — Whop could not be asked
// 401 signed out; 404 flag off.

// src/lib/settings/plan-view.ts
import type { NumenTier } from "@/lib/whop/config";
export interface SubscriptionInput {
  tier: NumenTier; status: string; isTrial: boolean; trialEndsAt: string | null; trialUsed: boolean;
  cancelAtPeriodEnd: boolean; currentPeriodEnd: string | null;
  usage: { limit: number | null; remaining: number | null; renewsAt: string | null } | null;
}
export type LiveInput = { membership: LiveMembership | null } | { unknown: true } | null; // null = not asked yet
export type PlanState = "renews" | "cancelled" | "pastDue" | "trial";
export type PlanCredits = { left: number; limit: number; resetsOn: string | null } | "unlimited" | null;
export type PlanView =
  | { kind: "none"; trialAvailable: boolean }
  | { kind: "plan"; name: string; price: string; state: PlanState; on: string | null; credits: PlanCredits };
export function planViewOf(sub: SubscriptionInput, live: LiveInput, now: Date): PlanView;
/** Delete may be offered: no plan, or a plan that will not bill again. */
export function canDelete(view: PlanView): boolean;
```

Rules for `planViewOf` (spec 2.1, 2.6):
- `tier === "free" && !isTrial` → `{ kind: "none", trialAvailable: !trialUsed }`. Never read `status` for this.
- The plan: `PLANS.find(p => p.id === tier)`; a tier with no plan (should not happen) → `{ kind: "none", trialAvailable: !trialUsed }`. `name` = plan name; `price` = `"$49 a month"` (`${plan.price} a month`).
- Cancelled: `live.membership.cancelAtPeriodEnd` when live has a membership, else `sub.cancelAtPeriodEnd`. Past due: live status `past_due`, else `sub.status === "past_due"`.
- `state`: pastDue → `"pastDue"`; else cancelled → `"cancelled"`; else `isTrial` → `"trial"`; else `"renews"`.
- `on` (a day word, "14 Dec"): trial (and not cancelled) → `trialEndsAt`; otherwise `live.membership.renewsAt ?? currentPeriodEnd`.
- `credits`: `usage === null` → `null`; `usage.limit === null` → `"unlimited"`; else `{ left: usage.remaining ?? 0, limit: usage.limit, resetsOn: day(usage.renewsAt) }`.
- `canDelete`: `view.kind === "none" || view.state === "cancelled"`.

`membershipRenews(m)`: `["active","trialing","past_due"].includes(m.status) && !m.cancelAtPeriodEnd`.

The plan-state route (spec 3.1):
```ts
// src/app/api/settings/plan-state/route.ts
import { NextResponse } from "next/server";
import { NEW_LOOK_ENABLED } from "@/lib/flags/new-look";
import { createClient } from "@/lib/supabase/server";
import { createServiceClient } from "@/lib/supabase/service";
import { fetchWhopMembership } from "@/lib/whop/membership";

export async function GET() {
  if (!NEW_LOOK_ENABLED) return NextResponse.json({ error: "not_found" }, { status: 404 });
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  const service = createServiceClient();
  const { data: row } = await service
    .from("user_subscriptions")
    .select("whop_membership_id, cancel_at_period_end, current_period_end")
    .eq("user_id", user.id)
    .maybeSingle();
  if (!row?.whop_membership_id) return NextResponse.json({ membership: null });
  let membership;
  try {
    membership = await fetchWhopMembership(row.whop_membership_id);
  } catch {
    return NextResponse.json({ unknown: true });
  }
  if (!membership) return NextResponse.json({ membership: null });
  // Catch the stored row up (the webhook never sends this flag), the caller's row only.
  if (membership.cancelAtPeriodEnd !== Boolean(row.cancel_at_period_end) || (membership.renewsAt && membership.renewsAt !== row.current_period_end)) {
    await service
      .from("user_subscriptions")
      .update({ cancel_at_period_end: membership.cancelAtPeriodEnd, ...(membership.renewsAt ? { current_period_end: membership.renewsAt } : {}) })
      .eq("user_id", user.id);
  }
  return NextResponse.json({ membership });
}
```

```ts
// src/lib/account/delete-account.ts
import type { SupabaseClient } from "@supabase/supabase-js";
import type { LiveMembership } from "@/lib/whop/membership";
export type DeleteOutcome =
  | { ok: true; removedFiles: number }
  | { ok: false; reason: "plan_renews"; renewsAt: string | null }
  | { ok: false; reason: "whop_unreachable" }
  | { ok: false; reason: "ledger" }
  | { ok: false; reason: "files_failed" }
  | { ok: false; reason: "delete_failed" };
export interface DeleteDeps {
  service: SupabaseClient;                                   // service-role client
  fetchMembership(id: string): Promise<LiveMembership | null>; // throws when Whop cannot be asked
  log(event: string, fields: Record<string, number | string | boolean>): void; // counts and step names only
}
export async function deleteAccount(userId: string, deps: DeleteDeps): Promise<DeleteOutcome>;

// POST /api/settings/account/delete  body { confirm: "DELETE" }
// 200 { deleted: true } | 400 { error: "confirm" } | 401 | 403 { error: "visitor" } (anonymous / sealed visitor)
// 409 { error: "plan_renews", renewsAt } | 409 { error: "ledger" } | 503 { error: "whop_unreachable" }
// 500 { error: "files_failed" } | 500 { error: "delete_failed" } | 404 flag off
```

`deleteAccount` steps, in this order (spec 3.2), stopping at the first refusal:
1. `service.from("user_subscriptions").select("whop_membership_id").eq("user_id", userId).maybeSingle()`. With an id: `fetchMembership(id)`; throws → `whop_unreachable`; a membership for which `membershipRenews` is true → `plan_renews` with its `renewsAt`. (A subscription read error → `whop_unreachable` is wrong; treat a read error as `delete_failed` before anything is touched.)
2. `service.from("wallet_transactions").select("id", { count: "exact", head: true }).eq("user_id", userId)`; count > 0 → `ledger`; an error → `delete_failed`.
3. Collect paths, then remove:
   - `avatars` and `videos`: list under `${userId}` (`storage.from(bucket).list(userId, { limit: 1000, offset })`, paging until a short page; nested folders are not expected under these prefixes — if an entry has no `id` (a folder), list inside it too).
   - `remix_blueprints` ids (`select("id").eq("user_id", userId)`) and `analysis_results` ids (same): for each id list `clips/{id}` and `filmstrips/{id}`.
   - Remove per bucket in chunks of 100 (`storage.from(bucket).remove(paths)`); any error → `files_failed` (log which bucket and how many were removed before it).
4. `service.auth.admin.deleteUser(userId)`; error → `delete_failed` (log that files were removed).
5. `{ ok: true, removedFiles }`.

The route:
```ts
// src/app/api/settings/account/delete/route.ts
import { NextResponse } from "next/server";
import { z } from "zod";
import { deleteAccount } from "@/lib/account/delete-account";
import { NEW_LOOK_ENABLED } from "@/lib/flags/new-look";
import { createLogger } from "@/lib/logger"; // use the app's logger (find its import in src/app/api/cron/sync-whop/route.ts)
import { isSealedVisitor } from "@/lib/onboarding/verdict-seal";
import { createClient } from "@/lib/supabase/server";
import { createServiceClient } from "@/lib/supabase/service";
import { fetchWhopMembership } from "@/lib/whop/membership";

const Body = z.object({ confirm: z.literal("DELETE") });
const STATUS = { plan_renews: 409, ledger: 409, whop_unreachable: 503, files_failed: 500, delete_failed: 500 } as const;

export async function POST(request: Request) {
  if (!NEW_LOOK_ENABLED) return NextResponse.json({ error: "not_found" }, { status: 404 });
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  if (user.is_anonymous || isSealedVisitor(user)) return NextResponse.json({ error: "visitor" }, { status: 403 });
  const parsed = Body.safeParse(await request.json().catch(() => null));
  if (!parsed.success) return NextResponse.json({ error: "confirm" }, { status: 400 });
  const log = createLogger({ module: "api/settings/account/delete" });
  const outcome = await deleteAccount(user.id, {
    service: createServiceClient(),
    fetchMembership: (id) => fetchWhopMembership(id),
    log: (event, fields) => log.info(event, fields),
  });
  if (outcome.ok) return NextResponse.json({ deleted: true });
  const body = outcome.reason === "plan_renews" ? { error: outcome.reason, renewsAt: outcome.renewsAt } : { error: outcome.reason };
  return NextResponse.json(body, { status: STATUS[outcome.reason] });
}
```
